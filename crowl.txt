module.exports.template = (data) => { 
    let warning = "";

    function getData() {
        return data[0].data;
    }

    // Проверка входных данных
    if (!data || !Array.isArray(data) || !data[0] || !data[0].data) {
        warning += "Invalid input data format;";
        return {
            template: {},
            warning: warning || "OK"
        };
    }

    const transformedData = getData();
    if (!transformedData || !Array.isArray(transformedData)) {
        warning += "Transformed data is not an array.";
        return {
            template: {},
            warning: warning || "OK"
        };
    }

    function validateMetaInf() {
        if (data[0]["Meta-Inf"]) {
            return data[0]["Meta-Inf"];
        }
        warning += "'Meta-Inf' is undefined or null or empty;";
        return {};
    }

    let metaInf = validateMetaInf();
    const categoriesLeft = [];
    const categoriesRight = [];

    function getField(metaInf, nameField, defaultValue) {
        if (typeof metaInf[nameField] === 'undefined' || metaInf[nameField] === null) {
            warning += "'" + nameField + "' is undefined or null, use default " + nameField + " = " + defaultValue + ";";
            return defaultValue;
        }
        return metaInf[nameField];
    }

    for (let i = 0; i < transformedData.length; i++) {
        categoriesLeft.push(transformedData[i].name);
        categoriesRight.push(transformedData[i].y);
    }

    const numericValues = categoriesRight.map(item => {
        const num = parseFloat(item.replace(/%/g, '').replace('+', ''));
        return isNaN(num) ? 0 : num;
    });

    // Находим максимальное абсолютное значение для симметричной шкалы
    const maxAbsValue = Math.max(
        Math.abs(Math.max(...numericValues)),
        Math.abs(Math.min(...numericValues))
    );

    const dataPos = numericValues.map(value => value > 0 ? value : 0);
    const dataNeg = numericValues.map(value => value <= 0 ? value : 0);

    return {
        template: {
            type: 'jpeg',
            options: {
                chart: {
                    inverted: true,
                    type: 'column',
                },
                title: {
                    text: '',
                    align: 'left'
                },
                xAxis: [{
                    categories: categoriesLeft,
                    reversed: true,
                    visible: false
                }, {
                    opposite: true,
                    reversed: false,
                    visible: false,
                    categories: categoriesRight,
                    linkedTo: 0,
                }],
                yAxis: {
                    max: maxAbsValue * 1.6,
                    min: -maxAbsValue * 1.6,
                    visible: false
                },
                series: [
                    {
                        name: 'leftShadow',
                        data: Array(categoriesRight.length).fill(-maxAbsValue * 1.6),
                        color: '#f2f2f2',
                        borderRadius: '80%'
                    }, {
                        name: 'rightShadow',
                        data: Array(categoriesRight.length).fill(maxAbsValue * 1.6),
                        color: '#f2f2f2',
                        borderRadius: '80%',
                    },
                    {
                        name: 'left',
                        data: dataNeg,
                        color: {
                            linearGradient: {x1: 0, x2: 0, y1: 0, y2: 1},
                            stops: [
                                [0, "#ffffff"],
                                [0.4, '#a6a6a6'],
                                [1, '#a6a6a6']
                            ]
                        },
                        borderRadius: '80%'
                    }, {
                        name: 'right',
                        data: dataPos,
                        color: {
                            linearGradient: {x1: 0, x2: 0, y1: 0, y2: 1},
                            stops: [
                                [0, '#2ca530'],
                                [0.6, '#2ca530'],
                                [1, "#ffffff"]
                            ]
                        },
                        borderRadius: '50%'
                    },
                ],
                legend: {
                    enabled: false
                },
                plotOptions: {
                    column: {
                        grouping: false,
                        shadow: false
