module.exports.template = (data) => {
    let warning = "";

    function getData() {
        return data[0]?.data;
    }

    // Удалена заглушка data = [...]
    // Используем переданные данные

    const transformedData = getData();
    if (!transformedData || !Array.isArray(transformedData)) {
        warning += "Transformed data is not an array.";
        return {
            template: {},
            warning: warning
        };
    }

    function validateMetaInf() {
        if (data[0]?.["Meta-Inf"]) {
            return data[0]["Meta-Inf"];
        }
        warning += "'Meta-Inf' is undefined or null or empty;";
        return {};
    }

    let metaInf = validateMetaInf();

    function getField(metaInf, nameField, defaultValue) {
        if (typeof metaInf[nameField] === 'undefined' || metaInf[nameField] === null) {
            warning += "'" + nameField + "' is undefined or null, use default " + nameField + " = " + defaultValue + ";";
            return defaultValue;
        }
        return metaInf[nameField];
    }

    const categoriesLeft = transformedData.map(item => item.name);
    const categoriesRight = transformedData.map(item => item.y);

    // Преобразование значений в числа с обработкой ошибок
    const numericValues = categoriesRight.map(item => {
        const num = parseFloat(item.replace(/%/g, '').replace(',', '.'));
        return isNaN(num) ? 0 : num;
    });

    if (numericValues.length === 0) {
        warning += "No valid numeric data;";
        return {
            template: {},
            warning: warning
        };
    }

    // Находим максимальное абсолютное значение
    const maxAbsoluteValue = Math.max(...numericValues.map(Math.abs));

    // Рассчитываем границы с коэффициентом 1.6
    const axisLimit = maxAbsoluteValue * 1.6;

    return {
        template: {
            type: 'jpeg',
            options: {
                chart: {
                    inverted: true,
                    type: 'column',
                },
                title: { text: '', align: 'left' },
                xAxis: [{
                    categories: categoriesLeft,
                    reversed: true,
                    visible: false
                }, {
                    opposite: true,
                    reversed: false,
                    visible: false,
                    categories: categoriesRight,
                    linkedTo: 0,
                }],
                yAxis: {
                    max: axisLimit,
                    min: -axisLimit,
                    visible: false
                },
                series: [
                    {
                        name: 'leftShadow',
                        data: Array(categoriesRight.length).fill(-axisLimit),
                        color: '#f2f2f2',
                        borderRadius: '80%'
                    }, {
                        name: 'rightShadow',
                        data: Array(categoriesRight.length).fill(axisLimit),
                        color: '#f2f2f2',
                        borderRadius: '80%',
                    },
                    {
                        name: 'left',
                        data: numericValues.map(v => v <= 0 ? v : 0),
                        color: { /* ... */ },
                        borderRadius: '80%'
                    }, {
                        name: 'right',
                        data: numericValues.map(v => v > 0 ? v : 0),
                        color: { /* ... */ },
                        borderRadius: '50%'
                    },
                ],
                legend: { enabled: false },
                plotOptions: { column: { grouping: false, shadow: false } },
                exporting: {
                    sourceWidth: getField(metaInf, "width", 5000),
                    sourceHeight: getField(metaInf, "height", 300),
                    allowHTML: true,
                    type: 'image/jpeg'
                },
                credits: { enabled: false }
            }
        },
        warning: warning || "OK"
    };
};
